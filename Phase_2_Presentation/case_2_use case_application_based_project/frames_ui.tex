\begin{frame}{2. System in Action -- Find Similar and the Feedback Loop}
\framesubtitle{Item-to-item retrieval through \texttt{POST /recommend/item} --- same heads, different anchor}
\vspace{0.5mm}
\begin{columns}[T]
\column{0.45\textwidth}
\vspace{-1mm}
\begin{center}
\screenshot{similar}{width=\linewidth,height=5.0cm,keepaspectratio}
\end{center}

\column{0.525\textwidth}
{\scriptsize\textbf{Same five heads, a different anchor}}
\begin{itemize}\tiny
\setlength{\itemsep}{2pt}
  \item The drawer calls \texttt{POST /recommend/item}: the query vector \emph{is} the anchor item's own 384-d embedding, so nothing is typed.
  \item \textbf{Captain America: Civil War, 0.913} $=0.681+0.126+0.063+0.021+0.021$.
  \item The same film was \textbf{\#2 at 0.852} in the text search. \textbf{Graph is $+0.126$ in both} --- it is a $(\mathit{user},\mathit{item})$ property, independent of the query. \textbf{Semantic moves $0.588\rightarrow0.681$} because the anchor changed from the typed string to the item vector.
\end{itemize}
\vspace{1mm}
{\scriptsize\textbf{Closing the loop}}
\begin{itemize}\tiny
\setlength{\itemsep}{2pt}
  \item Any card's like / skip / rate $\rightarrow$ \texttt{POST /interactions} $\rightarrow$ MySQL $\rightarrow$ EMA update ($\alpha=0.25$).
  \item The next query ranks from the updated vector --- adaptation is \textbf{one vector write}, never a retrain.
\end{itemize}
\end{columns}
\end{frame}
